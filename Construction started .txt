# Retry generation with safeguards to avoid division by zero and handle singularities.
import numpy as np
import matplotlib.pyplot as plt
import json, textwrap, os, csv

out_dir = "/mnt/data/construct_package"
os.makedirs(out_dir, exist_ok=True)

# Parameter grid
rhos = np.linspace(0.0, 1.0, 201)
as_ = np.linspace(-1.5, 1.5, 201)   # 'a' self-feedback
b = -0.1
mu = 0.0

R, A = np.meshgrid(rhos, as_)
denom = 1 - (1-R)*A
# compute x_star with masking for small denom
x_star = np.full_like(denom, np.nan, dtype=float)
mask_valid = np.abs(denom) > 1e-6
x_star[mask_valid] = ((1-R[mask_valid])*b + R[mask_valid]*mu) / denom[mask_valid]

# Stability proxy: require abs((1-rho)*a) < 1 for linear map local stability of iterative map
stability = np.abs((1-R)*A) < 1

# Clip extreme values for plotting aesthetics
x_star_plot = np.copy(x_star)
x_star_plot[np.isnan(x_star_plot)] = 0.0
x_star_plot = np.clip(x_star_plot, -5, 5)

# Heatmap of x*
fig1, ax1 = plt.subplots(figsize=(6,5))
c = ax1.pcolormesh(R, A, x_star_plot, shading='auto', cmap='RdYlBu')
ax1.set_xlabel('rho (ergodic coupling)')
ax1.set_ylabel('a (self-feedback)')
ax1.set_title('Equilibrium x* for linear reduction (b={:.2f}, mu={:.2f})'.format(b, mu))
fig1.colorbar(c, ax=ax1, label='x* (clipped)')
fig1.tight_layout()
fig1_path = os.path.join(out_dir, "heatmap_xstar.png")
fig1.savefig(fig1_path, dpi=150)

# Heatmap of stability mask
fig2, ax2 = plt.subplots(figsize=(6,5))
c2 = ax2.pcolormesh(R, A, stability, shading='auto', cmap='gray_r')
ax2.set_xlabel('rho')
ax2.set_ylabel('a')
ax2.set_title('Stability mask: |(1-rho)*a| < 1 (white = stable)')
fig2.colorbar(c2, ax=ax2, label='stable (1) / unstable (0)')
fig2.tight_layout()
fig2_path = os.path.join(out_dir, "stability_mask.png")
fig2.savefig(fig2_path, dpi=150)

# Slices: show x* vs rho for several a values, handle denom singularities
a_vals = [-1.0, -0.5, 0.0, 0.5, 1.0]
fig3, ax3 = plt.subplots(figsize=(8,4))
for av in a_vals:
    denom_slice = 1 - (1-rhos)*av
    xs = np.full_like(denom_slice, np.nan)
    mask_ok = np.abs(denom_slice) > 1e-6
    xs[mask_ok] = ((1-rhos[mask_ok])*b + rhos[mask_ok]*mu) / denom_slice[mask_ok]
    ax3.plot(rhos, xs, label=f'a={av}')
ax3.set_xlabel('rho')
ax3.set_ylabel('x*')
ax3.set_title('x* vs rho for selected a values')
ax3.legend()
ax3.grid(True, linestyle='--', linewidth=0.4)
fig3_path = os.path.join(out_dir, "slices_xstar_vs_rho.png")
fig3.savefig(fig3_path, dpi=150, bbox_inches='tight')

# Save a CSV-style table of some sample computed x*
sample_points = []
for av in [-1.0, -0.5, 0.0, 0.5, 1.0]:
    for rv in [0.0, 0.25, 0.5, 0.75, 1.0]:
        denom_s = 1 - (1-rv)*av
        if abs(denom_s) < 1e-6:
            xs = float('nan')
            stable = False
        else:
            xs = ((1-rv)*b + rv*mu) / denom_s
            stable = abs((1-rv)*av) < 1
        sample_points.append((rv, av, xs, stable))

csv_path = os.path.join(out_dir, "sample_xstar_table.csv")
with open(csv_path, 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['rho','a','x_star','stable'])
    for row in sample_points:
        writer.writerow([f"{row[0]:.2f}", f"{row[1]:.2f}", f"{row[2] if not np.isnan(row[2]) else 'NaN'}", int(row[3])])

# Create Streamlit app code (build-ready) as a single Python file
streamlit_code = r'''
# Streamlit app: Construct "God Equation" tuner
# Usage: pip install streamlit numpy matplotlib
# Run: streamlit run streamlit_app.py

import streamlit as st
import numpy as np
import matplotlib.pyplot as plt

st.set_page_config(layout="wide")
st.title("Construct: God Equation Tuner — Scientific Mode")

# Sidebar controls
st.sidebar.header("Parameters (scientific)")
a = st.sidebar.slider("a (self-feedback)", -1.5, 1.5, 0.6, 0.01)
b = st.sidebar.slider("b (baseline bias)", -1.0, 1.0, -0.1, 0.01)
rho = st.sidebar.slider("rho (ergodic coupling)", 0.0, 1.0, 0.4, 0.01)
mu = st.sidebar.slider("mu (ensemble mean)", -2.0, 2.0, 0.0, 0.01)
show_time = st.sidebar.checkbox("Show time-series simulation", True)
reset_strength = st.sidebar.slider("Reset strength (s)", 0.0, 1.0, 0.8, 0.01)
reset_times = st.sidebar.multiselect("Reset iterations (comma-separated)", [50,100,150,200,300], default=[100,250])

# Compute fixed point (analytic)
denom = 1 - (1-rho)*a
if abs(denom) < 1e-6:
    x_star = float('nan')
    stability = False
else:
    x_star = ((1-rho)*b + rho*mu) / denom
    stability = abs((1-rho)*a) < 1

col1, col2 = st.columns([1,1])

with col1:
    st.subheader("Analytic fixed point")
    st.write("x* = ((1-rho)*b + rho*mu) / (1 - (1-rho)*a)")
    st.metric("x*", f"{x_star:.6f}" if (not np.isnan(x_star)) else "undefined")
    st.write("Stability (| (1-rho)*a | < 1):", "✅ Stable" if stability else "🔴 Unstable / Near singular")

    # Show phase portrait slice
    xs = np.linspace(-2,2,400)
    map_vals = (1-rho)* (a*xs + b) + rho*mu
    fig, ax = plt.subplots(figsize=(5,4))
    ax.plot(xs, map_vals, label='map: x_{n+1}')
    ax.plot(xs, xs, '--', label='identity line')
    ax.set_xlim(-2,2); ax.set_ylim(-2,2)
    ax.set_title(f"Phase slice (a={a:.2f}, rho={rho:.2f})")
    ax.legend()
    st.pyplot(fig)

with col2:
    st.subheader("Parameter impacts (slices)")
    rhos = np.linspace(0,1,201)
    denom_s = 1 - (1-rhos)*0.5  # example slice, show generic behavior
    xs = ((1-rhos)*b + rhos*mu) / np.where(np.abs(denom_s)>1e-6, denom_s, np.nan)
    fig2, ax2 = plt.subplots(figsize=(6,2))
    ax2.plot(rhos, xs, label=f'a=0.5 example')
    ax2.set_xlabel('rho'); ax2.set_ylabel('x*')
    ax2.grid(True, linestyle='--', linewidth=0.4)
    st.pyplot(fig2)

# Time-series simulation
if show_time:
    T = 400
    x = np.zeros(T)
    x[0] = -0.8
    sigma = 0.02
    for n in range(T-1):
        x[n+1] = (1-rho)*(a*x[n] + b) + rho*mu + sigma*np.random.randn()
        if (n+1) in reset_times:
            s = reset_strength
            x[n+1] = (1-s)*x[n+1] + s*mu
    st.subheader("Time-series simulation (linear local map)")
    figt, axt = plt.subplots(figsize=(8,3))
    axt.plot(x); axt.set_xlabel("iteration n"); axt.set_ylabel("x_n")
    for rt in reset_times:
        axt.axvline(rt, linestyle='--', alpha=0.7)
    st.pyplot(figt)

st.sidebar.markdown("---")
st.sidebar.markdown("Export config: use the 'Share' button or copy parameter values.")
st.write("Notes: This is a scientific reduction (linear local map). For the full operator model see the technical manual.")
'''

streamlit_path = os.path.join(out_dir, "streamlit_app.py")
with open(streamlit_path, 'w') as f:
    f.write(streamlit_code)
